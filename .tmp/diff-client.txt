    feat(dsh-plugin): open capability groups from the session strip
    
    The group window used to hang off a workspace row's `...` menu, so the
    directory it switched belonged to whichever row was right-clicked. A session
    already owns one workspace, so the entry moves onto the session itself: an
    icon in the composer's bottom-left strip opens the same window for the
    workspace that owns that session, and the window writes
    `<workspace>/.hnflow/stack.json` there.
    
    - The button is registered on `conversation.input.left` next to the Agent
      picker and the inventory preview. It is not gated on the HNFlow mode, so a
      session can switch groups before it switches its Agent.
    - The workspace is the record whose `sessionIds` contain this session; the
      session snapshot's `cwd` is only a fallback for a host without workspace
      records, and it also covers a session working in a subdirectory.
    - The `contextMenus` contribution and its host requirement are gone, so the
      entry no longer depends on the DSH version that ships the workspace menu.
    - The window header names the exact directory it writes to, and a body hint
      repeats that path; the overlay registration is unchanged.
    
    Tests: the client contract drops the workspace-menu case and pins the strip
    button, the workspace resolution order and the empty menu list;
    `node test/run.mjs` and `python -m pytest tests -q` both pass.

+/**
+ * Workspace the composer strip writes capability-group switches to.
+ *
+ * The window used to open from a workspace row, so the directory came from the
+ * clicked row. A session owns exactly one workspace: the workspace record lists
+ * its `sessionIds`, and that path, not the session's own `cwd`, is where
+ * `.hnflow/stack.json` lives. The session snapshot is only a fallback for a
+ * host that does not publish workspace records yet.
+ * @param {any} sessions - sessions snapshot.
+ * @param {any} workspaces - workspaces snapshot.
+ * @param {string} sessionId - window session id.
+ * @param {any} sessionsService - injectable client sessions service.
+ * @returns {string} absolute directory, or an empty string.
+ */
+function sessionWorkspace(sessions, workspaces, sessionId, sessionsService) {
+  if (typeof sessionId !== "string" || sessionId.length === 0) return "";
+  for (const record of workspaceRecords(workspaces)) {
+    if (record === null || typeof record !== "object") continue;
+    const owned = Array.isArray(record.sessionIds) && record.sessionIds.includes(sessionId);
+    if (owned && typeof record.path === "string" && record.path.length > 0) return record.path;
+  }
+  const projected = inventoryCwdHint(sessions, workspaces, sessionId);
+  if (projected.length > 0) return projected;
+  return sessionCwdHint(sessionsService, sessionId);
+}
+
+/**
+ * Last path segment of a directory, for the window title.
+ *
+ * Both separators are accepted because the client reports Windows paths with
+ * either one depending on where the value came from.
+ * @param {string} directory - absolute directory.
+ * @returns {string} bare name, or an empty string.
+ */
+function workspaceBaseName(directory) {
+  const text = typeof directory === "string" ? directory.trim().replace(/[\\/]+$/u, "") : "";
+  if (text.length === 0) return "";
+  const cut = Math.max(text.lastIndexOf("/"), text.lastIndexOf("\\"));
+  return cut >= 0 ? text.slice(cut + 1) : text;
+}
